// Some working proof examples

lemma d_neg: P -> ~~P
proof (rule Imp_I) {
    assume P
    show ~~P
    proof (rule Falsity_E) {
        assume ~P
        from ~P and P show F by (rule Neg_E)
    }
}

lemma imp_inv: ((P -> Q) & ~Q) -> ~P
proof (rule Imp_I) {
	assume (P -> Q) & ~Q
	show ~P
    proof (rule Neg_I) {
        assume P
		have P -> Q by (rule Con_E1)
		have Q by (rule Imp_E)
        have ~Q by (rule Con_E2)
        show F by (rule Neg_E)
	}
}

lemma b-iimp_eqv: (P <-> Q) <-> (Q <-> P)
proof (rule Iff_I) {
	assume P <-> Q
    show Q <-> P
    proof (rule Iff_I) {
        have P ==> Q by (rule Iff_E1)
        have Q ==> P by (rule Iff_E2)
    }
    assume Q <-> P
	show P <-> Q
    proof (rule Iff_I) {
        have Q ==> P by (rule Iff_E1)
        have P ==> Q by (rule Iff_E2)
    }
}

lemma classical: (~P ==> P) ==> P
proof this {
	assume ~P ==> P
	have P ==> P by this
	have P | ~P by (rule LEM)
	show P by (rule Dis_E)
}

lemma Clavius: (~P -> P) -> P
proof (rule Imp_I) {
	assume ~P -> P
	show P
	proof (rule classical) {
		assume ~P
		show P by (rule Imp_E)
	}
}

lemma Peirce: ((P -> Q) -> P) -> P
proof (rule Imp_I) {
	assume (P -> Q) -> P
	show P
	proof (rule classical) {
		assume ~P
		have P -> Q
		proof (rule Imp_I) {
			assume P
			show Q by (rule Imp_E)
		}
		show P by (rule Imp_E)
	}
}

lemma dneg: (~~p) -> p
proof (rule Imp_I) {
    assume ~~p
    show p
    proof (rule Ccontr) {
        assume ~p
        from ~~p and ~p have F by Neg_E
        from ~p and F show (~p) -> F by Imp_I
    }
}
